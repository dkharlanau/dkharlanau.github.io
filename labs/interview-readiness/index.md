  <section class="research-canvas__method" data-reveal>
    <div><p class="research-canvas__eyebrow">Interview loop</p><h2>Refresh, explain, pressure-test, review.</h2></div>
    <ol>
      <li><span>01</span><strong>Refresh</strong><p>Open the linked Lab page and rebuild the process or architecture model.</p></li>
      <li><span>02</span><strong>Explain</strong><p>Answer one question without notes. Keep the business goal and ownership visible.</p></li>
      <li><span>03</span><strong>Pressure-test</strong><p>Use Interview Mode or the Assessment Lab. Let follow-up questions expose weak boundaries.</p></li>
      <li><span>04</span><strong>Review</strong><p>Update the roadmap and one project story. Your next session should change because of the result.</p></li>
    </ol>
  </section>

<script>
(() => {
  'use strict';
  const IR = window.InterviewReadiness;
  if (!IR) return;
  const overall = document.getElementById('ir-overall');
  const counts = document.getElementById('ir-counts');
  const tracks = document.getElementById('ir-track-summary');
  const daily = document.getElementById('ir-daily-plan');

  function render() {
    const data = IR.summary();
    overall.textContent = data.overall + '%';
    const reviewed = IR.TOPICS.length - data.statuses['not-reviewed'];
    counts.textContent = reviewed ? `${reviewed} of ${IR.TOPICS.length} topics touched.` : 'No topics reviewed yet.';

    tracks.replaceChildren();
    Object.entries(data.tracks).forEach(([id,track]) => {
      const card = document.createElement('article');
      card.className = 'ir-card ir-track';
      card.innerHTML = `<div class="ir-track__head"><strong>${track.label}</strong><span>${track.readiness}%</span></div><div class="ir-track__bar" aria-hidden="true"><span style="width:${track.readiness}%"></span></div><small>${track.count} topics</small>`;
      tracks.appendChild(card);
    });

    daily.replaceChildren();
    IR.dailyPlan(4).forEach((item,index) => {
      const row = document.createElement('article');
      row.className = 'ir-daily__item';
      const action = item.status.score === 0 ? 'Refresh' : item.status.score === 1 ? 'Explain' : 'Defend';
      row.innerHTML = `<span class="ir-daily__time">${item.minutes} min</span><div><strong>${action}: ${item.title}</strong><small>${item.why}</small></div><a class="ir-link" href="${item.href}">Open source <span class="material-symbols-outlined" aria-hidden="true">arrow_forward</span></a>`;
      daily.appendChild(row);
    });
  }
  render();
  window.addEventListener('interview-readiness-change', render);
})();
</script>
